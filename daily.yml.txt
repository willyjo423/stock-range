name: Daily

# Publishes today's ranges and grades the ones whose windows have closed.
on:
  schedule:
    # 22:30 UTC on weekdays - after the US close, before the next open.
    - cron: "30 22 * * 1-5"
  workflow_dispatch:

permissions:
  contents: write

concurrency:
  group: daily
  cancel-in-progress: false

jobs:
  forecast:
    runs-on: ubuntu-latest
    timeout-minutes: 60

    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"
          cache: pip

      - name: Install
        run: pip install -r requirements.txt

      - name: Restore data cache
        uses: actions/cache@v4
        with:
          path: data/cache
          key: vol-daily-${{ github.run_id }}
          restore-keys: |
            vol-daily-
            vol-cache-

      - name: Offline checks
        run: |
          python test_signals.py
          python test_options.py

      - name: Forecast
        run: python daily.py

      - name: Options view
        id: options
        # Needs the forecasts from the step above, and fetches the chains
        # itself. Never let it take the ranges down with it.
        continue-on-error: true
        run: python options_view.py

      - name: Say whether the options page built
        # continue-on-error means the run goes green either way, so without
        # this the only symptom of a failure is a 404 on a link nobody is
        # watching. It writes a page explaining itself now; this puts the
        # outcome where it can be seen without opening the log.
        if: always()
        run: |
          if [ "${{ steps.options.outcome }}" = "success" ]; then
            echo "### Options page: built" >> $GITHUB_STEP_SUMMARY
          else
            echo "### Options page: FELL BACK TO A PLACEHOLDER" \
              >> $GITHUB_STEP_SUMMARY
            echo "The page exists and says why, but nothing was priced." \
              >> $GITHUB_STEP_SUMMARY
          fi
          test -f docs/options.html \
            && echo "docs/options.html present ($(wc -c < docs/options.html) bytes)" \
            || echo "docs/options.html MISSING - the link will 404"

      - name: Grade ranges whose windows have closed
        # Never let the scoreboard take the forecasts down with it.
        continue-on-error: true
        run: python track.py --write

      - name: Publish
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add -f docs/ data/forecasts/ data/implied/
          git diff --staged --quiet && echo "nothing changed" && exit 0
          git commit -m "Ranges: $(date -u +%Y-%m-%d)"
          git pull --rebase --autostash origin main || true
          git push
