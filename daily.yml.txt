name: Daily

# Publishes today's ranges and grades the ones whose windows have closed.
on:
  schedule:
    # 22:30 UTC on weekdays - after the US close, before the next open.
    - cron: "30 22 * * 1-5"
  workflow_dispatch:

permissions:
  contents: write

concurrency:
  group: daily
  cancel-in-progress: false

jobs:
  forecast:
    runs-on: ubuntu-latest
    timeout-minutes: 60

    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"
          cache: pip

      - name: Install
        run: pip install -r requirements.txt

      - name: Restore data cache
        uses: actions/cache@v4
        with:
          path: data/cache
          key: vol-daily-${{ github.run_id }}
          restore-keys: |
            vol-daily-
            vol-cache-

      - name: Offline checks
        run: |
          python test_signals.py
          python test_options.py

      - name: Forecast
        run: python daily.py

      - name: Options view
        id: options
        # Needs the forecasts from the step above, and fetches the chains
        # itself. Never let it take the ranges down with it.
        continue-on-error: true
        run: python options_view.py

      - name: Guarantee the options page exists
        # The last line of defence, and the only one that does not depend on
        # Python running at all. options_view.py writes its own explanation
        # when the BUILD fails - but if the module fails to import, or the
        # interpreter dies before main(), nothing runs and the link 404s.
        # This is shell, it runs whatever happened above, and a page that
        # says "not built" is strictly better than GitHub's error screen.
        if: always()
        run: |
          mkdir -p docs
          if [ ! -s docs/options.html ]; then
            echo "options.html missing or empty - writing a stand-in"
            cat > docs/options.html <<'HTML'
          <!doctype html><html lang="en"><head><meta charset="utf-8">
          <meta name="viewport" content="width=device-width,initial-scale=1">
          <title>Options view</title><style>
          body{margin:0;background:#f6f7f9;color:#14171c;font:15px/1.6
          -apple-system,BlinkMacSystemFont,'Segoe UI',Roboto,sans-serif}
          @media(prefers-color-scheme:dark){body{background:#0f1115;color:#e8eaed}
          .note{background:#171a21;border-color:#252a34}a{color:#4ea1ff}}
          .wrap{max-width:680px;margin:0 auto;padding:40px 16px}
          h1{font-size:22px;margin:0 0 18px}
          .note{background:#fff;border:1px solid #e3e6ea;border-left:3px solid
          #d29922;border-radius:8px;padding:16px 18px;font-size:14px}
          footer{margin-top:28px;font-size:13px}</style></head><body>
          <div class="wrap"><h1>Options view</h1>
          <div class="note"><b>This page did not build on the last run.</b><br>
          The options comparison needs live chains and could not produce one.
          The ranges themselves are unaffected. Check the Daily run log for the
          Options view step, and this page will rebuild on the next run.</div>
          <footer><a href="./">Ranges</a> &middot;
          <a href="./results.html">How the ranges have scored</a> &middot;
          <a href="./flow.html">Options flow</a></footer>
          </div></body></html>
          HTML
          fi
          ls -l docs/options.html

      - name: Say whether the options page built
        # continue-on-error means the run goes green either way, so without
        # this the only symptom of a failure is a 404 on a link nobody is
        # watching. It writes a page explaining itself now; this puts the
        # outcome where it can be seen without opening the log.
        if: always()
        run: |
          if [ "${{ steps.options.outcome }}" = "success" ]; then
            echo "### Options page: built" >> $GITHUB_STEP_SUMMARY
          else
            echo "### Options page: FELL BACK TO A PLACEHOLDER" \
              >> $GITHUB_STEP_SUMMARY
            echo "The page exists and says why, but nothing was priced." \
              >> $GITHUB_STEP_SUMMARY
          fi
          test -f docs/options.html \
            && echo "docs/options.html present ($(wc -c < docs/options.html) bytes)" \
            || echo "docs/options.html MISSING - the link will 404"

      - name: Grade ranges whose windows have closed
        # Never let the scoreboard take the forecasts down with it.
        continue-on-error: true
        run: python track.py --write

      - name: Publish
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add -f docs/ data/forecasts/ data/implied/
          git diff --staged --quiet && echo "nothing changed" && exit 0
          git commit -m "Ranges: $(date -u +%Y-%m-%d)"
          git pull --rebase --autostash origin main || true
          git push
